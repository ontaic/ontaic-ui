How It Works

From Python code to blazing-fast UI in 3 steps

1

Write Python

from ontaic import *

class Counter(Component):
    count: int = 0

    def render(self):
        return Container(
            Heading(f"Count: {self.count}"),
            Button("Increment", on_click=self.count.add(1))
        )
2

Compile to Schema

{
  "type": "container",
  "children": [
    {
      "type": "heading",
      "text": "Count: {count}"
    },
    {
      "type": "button",
      "text": "Increment",
      "onClick": "state.count++"
    }
  ]
}
3

Run in WASM

// Rust WASM patches DOM directly
fn patch_dom(op: PatchOp) {
    match op {
        PatchOp::SetText { id, text } => {
            document.get_element_by_id(&id)
                .set_text_content(&text);
        }
        PatchOp::SetAttr { id, key, val } => {
            // Direct DOM manipulation
        }
    }
}

Quick Start

Get up and running in 2 minutes

1. Install the CLI

cargo install ontaic-cli

2. Create a project

ontaic init my-app
cd my-app

3. Add a component

ontaic add counter

4. Start development

ontaic dev
