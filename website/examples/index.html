Counter

View Source Live Demo

A simple counter demonstrating state management and event handling.

from ontaic import *

class Counter(Component):
    count: int = 0
    
    def render(self):
        return Container(
            Heading(f"Count: {self.count}"),
            Flex(
                Button("+", on_click=self.count.add(1)),
                Button("-", on_click=self.count.sub(1)),
                Button("Reset", on_click=self.count.set(0)),
                gap="0.5rem"
            ),
            center=True,
            padding="2rem"
        )

Todo App

View Source Live Demo

A full-featured todo app with routing, forms, and local storage.

from ontaic import *

class TodoApp(Component):
    todos: list = []
    new_todo: str = ""
    filter: str = "all"
    
    def add_todo(self):
        if self.new_todo.strip():
            self.todos.append({
                "id": len(self.todos),
                "text": self.new_todo,
                "done": False
            })
            self.new_todo = ""
    
    def toggle_todo(self, id):
        for todo in self.todos:
            if todo["id"] == id:
                todo["done"] = not todo["done"]
    
    def render(self):
        return Container(
            Heading("Todo List"),
            Flex(
                Input(
                    value=self.new_todo,
                    on_change=self.new_todo.set,
                    placeholder="Add a todo..."
                ),
                Button("Add", on_click=self.add_todo),
                gap="0.5rem"
            ),
            ForEach(
                self.todos,
                lambda todo: Flex(
                    Checkbox(
                        checked=todo["done"],
                        on_change=lambda: self.toggle_todo(todo["id"])
                    ),
                    Text(todo["text"]),
                    gap="0.5rem"
                )
            ),
            spacing="1rem"
        )

Forms Example

View Source

Comprehensive form example with validation and多种 input types.

from ontaic import *

class FormExample(Component):
    name: str = ""
    email: str = ""
    password: str = ""
    birthday: str = ""
    color: str = "#3b82f6"
    agreed: bool = False
    
    def render(self):
        return Container(
            Form(
                FormField("Name", Input(value=self.name, on_change=self.name.set)),
                FormField("Email", Input(value=self.email, type="email")),
                FormField("Password", Input(value=self.password, type="password")),
                FormField("Birthday", DatePicker(value=self.birthday)),
                FormField("Favorite Color", ColorPicker(value=self.color)),
                FormField("I agree", Checkbox(checked=self.agreed, on_change=self.agreed.toggle)),
                Button("Submit", type="submit"),
                spacing="1rem"
            ),
            max_width="400px",
            margin="0 auto"
        )

Data Grid Example

View Source

Advanced data grid with sorting, filtering, and pagination.

from ontaic import *

columns = [
    grid_column("name", "Name", sortable=True),
    grid_column("email", "Email"),
    grid_column("role", "Role"),
    grid_column("status", "Status"),
]

data = [
    {"name": "John Doe", "email": "john@example.com", "role": "Admin", "status": "Active"},
    {"name": "Jane Smith", "email": "jane@example.com", "role": "User", "status": "Active"},
    {"name": "Bob Wilson", "email": "bob@example.com", "role": "User", "status": "Inactive"},
]

class UsersGrid(Component):
    def render(self):
        return Container(
            Heading("Users"),
            DataGrid(
                columns=columns,
                data=data,
                sortable=True,
                filterable=True,
                pagination=True,
                page_size=10
            )
        )
